# Auto-generated by ADMS Converter Agent — review before promoting to prod
from pyspark.sql import functions as F

df_event = spark.read.table("oms_bronze.outage_event")
df_cust  = spark.read.table("oms_bronze.outage_cust")

silver = (
    df_event
    .join(df_cust, "outage_id", "left")
    .select(
        F.col("OUTAGE_ID").alias("outage_id"),
        F.col("DEVICE_ID").alias("device_id"),
        F.col("REST_TS").alias("restore_ts"),
        F.col("CUST_CNT").alias("customer_count"),  # confidence: assumed
    )
)

silver.write.mode("merge").saveAsTable("adms.silver.outage_event")

Why was CUST_CNT marked "Assumed" instead of "Converted"?
No exact match existed in the ADMS data dictionary for CUST_CNT. I matched it by semantic similarity to customer_count, a field used identically in 3 other already-converted objects — high confidence, but not a documented 1:1 mapping, so it's flagged Assumed rather than Converted until an engineer confirms it.
